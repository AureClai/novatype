Mise en forme

Personnalisez l'apparence de vos documents avec les regles de style NovaType.

Configuration de la page

#set page(
  paper: "a4",                  // Format: a4, letter, a5...
  margin: (
    top: 2.5cm,
    bottom: 2cm,
    left: 2.5cm,
    right: 2cm
  ),
  header: [
    #align(right)[Mon Document]
    #line(length: 100%)
  ],
  footer: [
    #align(center)[
      Page #counter(page).display()
    ]
  ],
  numbering: "1"               // Numerotation des pages
)

Formats de page disponibles

Format Dimensions Usage
"a4"210 x 297 mmStandard europeen
"letter"8.5 x 11 inStandard americain
"a5"148 x 210 mmPetit format
"presentation-16-9"16:9Slides

Configuration du texte

#set text(
  font: "Arial",           // Police principale
  size: 11pt,              // Taille de base
  fill: rgb("#333333"),   // Couleur du texte
  lang: "fr",              // Langue (cesure, guillemets)
  weight: "regular",       // Graisse: thin, regular, bold
  style: "normal"          // Style: normal, italic, oblique
)

Polices populaires

// Serif (documents formels)
#set text(font: "Times New Roman")
#set text(font: "Georgia")
#set text(font: "Garamond")

// Sans-serif (documents modernes)
#set text(font: "Arial")
#set text(font: "Helvetica")
#set text(font: "Calibri")

// Monospace (code)
#set raw(font: "JetBrains Mono")

Configuration des paragraphes

#set par(
  justify: true,           // Justification
  leading: 0.65em,         // Interligne
  first-line-indent: 1em,  // Retrait premiere ligne
  spacing: 1.2em           // Espace entre paragraphes
)

Styles des titres

// Numerotation
#set heading(numbering: "1.1")

// Style personnalise
#show heading.where(level: 1): it => {
  set text(size: 18pt, weight: "bold", fill: rgb("#2563eb"))
  v(1em)
  block(it)
  v(0.3em)
  line(length: 100%, stroke: 2pt + rgb("#2563eb"))
  v(0.8em)
}

#show heading.where(level: 2): it => {
  set text(size: 14pt, weight: "bold")
  v(0.8em)
  block(it)
  v(0.4em)
}

Couleurs

// Definition de couleurs
#let primary = rgb("#2563eb")
#let secondary = rgb("#64748b")
#let accent = rgb("#059669")

// Utilisation
#text(fill: primary)[Texte en bleu]

// Autres formats
rgb(255, 0, 0)          // RGB
rgb("#ff0000")          // Hex
cmyk(0%, 100%, 100%, 0%) // CMYK
luma(50%)               // Niveau de gris

Blocs et encadres

// Bloc simple
#block(
  fill: rgb("#f1f5f9"),
  inset: 1em,
  radius: 4pt,
  width: 100%
)[
  Contenu du bloc avec fond colore et coins arrondis.
]

// Encadre avec bordure
#block(
  stroke: 1pt + rgb("#2563eb"),
  inset: 1em,
  radius: 4pt
)[
  Contenu encadre avec bordure bleue.
]

// Note ou avertissement
#block(
  fill: rgb("#fef3c7"),
  stroke: (left: 4pt + rgb("#f59e0b")),
  inset: 1em
)[
  *Attention :* Ceci est un avertissement important.
]

Lignes et separateurs

// Ligne horizontale
#line(length: 100%)

// Ligne avec style
#line(
  length: 100%,
  stroke: 2pt + rgb("#2563eb")
)

// Ligne pointillee
#line(
  length: 100%,
  stroke: (dash: "dashed")
)

Colonnes

// Document en deux colonnes
#set page(columns: 2)

// Section en colonnes
#columns(2, gutter: 1cm)[
  Contenu de la premiere colonne...

  #colbreak()

  Contenu de la seconde colonne...
]

Variables et fonctions personnalisees

// Variables
#let titre = "Mon Document"
#let auteur = "Jean Dupont"

// Fonction personnalisee
#let note(body) = block(
  fill: rgb("#e0f2fe"),
  inset: 1em,
  radius: 4pt,
  width: 100%
)[
  #text(weight: "bold")[Note :] #body
]

// Utilisation
#note[Ceci est une note importante.]
set vs show

#set definit des valeurs par defaut. #show transforme l'apparence des elements existants.
